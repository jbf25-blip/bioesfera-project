    <script>
        const personagem = document.getElementById('personagem');
        const velocidade = 25;
        let posX = 200;
        let posY = 230;

        document.addEventListener('keydown', (e) => {
            if (e.key === 'ArrowUp') posY -= velocidade;
            if (e.key === 'ArrowDown') posY += velocidade;
            if (e.key === 'ArrowRight') {
                posX += velocidade;
                personagem.style.transform = 'scaleX(1)'; 
            }
            if (e.key === 'ArrowLeft') {
                posX -= velocidade;
                personagem.style.transform = 'scaleX(-1)'; 
            }

            // Limites de tela
            posY = Math.max(0, Math.min(460, posY));
            posX = Math.max(0, Math.min(760, posX));

            personagem.style.top = posY + 'px';
            personagem.style.left = posX + 'px';
        });
    </script>
